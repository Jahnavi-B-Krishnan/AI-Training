CREATE OR REPLACE PROCEDURE increase_service_cost(
    p_percentage NUMERIC
)
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE service_items
    SET cost = cost * (1 + p_percentage / 100.0);
END;
$$;

CALL increase_service_cost(10);
